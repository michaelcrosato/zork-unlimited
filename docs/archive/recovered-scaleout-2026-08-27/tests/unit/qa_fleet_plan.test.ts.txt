import { describe, expect, it } from "vitest";
import {
  buildPersonaCycle,
  DEFAULT_PERSONA_WEIGHTS,
  extractExitInterview,
  mulberry32,
  parseQaArgs,
  planQaRuns,
  QA_TIERS,
  recoverQaArgs,
  summarizeQaReports,
  // @ts-expect-error — plain .mjs module without type declarations
} from "../../blind-tester/qa/qa-fleet.mjs";

describe("qa fleet planning", () => {
  it("is deterministic: same inputs produce the identical roster", () => {
    const a = planQaRuns({ count: 25, seedBase: 900500, thinkRate: 0.01 });
    const b = planQaRuns({ count: 25, seedBase: 900500, thinkRate: 0.01 });
    expect(a).toEqual(b);
  });

  it("assigns contiguous seeds from the seed base", () => {
    const plan = planQaRuns({ count: 5, seedBase: 1234, thinkRate: 0 });
    expect(plan.map((r: { seed: number }) => r.seed)).toEqual([1234, 1235, 1236, 1237, 1238]);
  });

  it("pins tier extremes: rate 0 is all cheap, rate 1 is all think", () => {
    const cheap = planQaRuns({ count: 12, seedBase: 42, thinkRate: 0 });
    const think = planQaRuns({ count: 12, seedBase: 42, thinkRate: 1 });
    expect(cheap.every((r: { tier: string }) => r.tier === "cheap")).toBe(true);
    expect(think.every((r: { tier: string }) => r.tier === "think")).toBe(true);
    expect(cheap[0].model).toBe(QA_TIERS.cheap.model);
    expect(think[0].model).toBe(QA_TIERS.think.model);
  });

  it("samples roughly 1-in-100 max-think runs at the default rate", () => {
    const plan = planQaRuns({ count: 1000, seedBase: 900500, thinkRate: 0.01 });
    const thinkRuns = plan.filter((r: { tier: string }) => r.tier === "think");
    // Deterministic for this seed base; loose bounds document the intent.
    expect(thinkRuns.length).toBeGreaterThan(0);
    expect(thinkRuns.length).toBeLessThan(40);
    for (const run of thinkRuns) {
      expect(run.model).toBe(QA_TIERS.think.model);
      expect(run.effort).toBe("xhigh");
    }
  });

  it("cycles every persona with the configured weighting", () => {
    const cycle = buildPersonaCycle(DEFAULT_PERSONA_WEIGHTS);
    const plan = planQaRuns({ count: cycle.length, seedBase: 7, thinkRate: 0 });
    const counts = new Map<string, number>();
    for (const run of plan as { persona: string }[]) {
      counts.set(run.persona, (counts.get(run.persona) ?? 0) + 1);
    }
    for (const [name, weight] of DEFAULT_PERSONA_WEIGHTS as [string, number][]) {
      expect(counts.get(name)).toBe(weight);
    }
    // Critical-leaning: critic+skeptic+impatient+breaker dominate the cycle.
    const critical =
      (counts.get("critic") ?? 0) +
      (counts.get("skeptic") ?? 0) +
      (counts.get("impatient") ?? 0) +
      (counts.get("breaker") ?? 0);
    expect(critical * 2).toBeGreaterThan(cycle.length);
  });

  it("rejects invalid plans", () => {
    expect(() => planQaRuns({ count: 0, seedBase: 1, thinkRate: 0 })).toThrow(/--count/);
    expect(() => planQaRuns({ count: 1, seedBase: -1, thinkRate: 0 })).toThrow(/--seed-base/);
    expect(() => planQaRuns({ count: 1, seedBase: 1, thinkRate: 2 })).toThrow(/--think-rate/);
    expect(() => buildPersonaCycle([["critic", 0]])).toThrow(/positive integer/);
  });

  it("parses equals-form flags and recovers npm-eaten flags", () => {
    const parsed = parseQaArgs(["--count=20", "--seed-base=900500", "--dry-run"]);
    expect(parsed.count).toBe(20);
    expect(parsed.seedBase).toBe(900500);
    expect(parsed.dryRun).toBe(true);
    // npm recorded --count=20 as config; argv kept only --dry-run.
    expect(recoverQaArgs(["--dry-run"], { npm_config_count: "20" })).toEqual([
      "--dry-run",
      "--count",
      "20",
    ]);
    // Space-separated flag: npm records true and orphans the value.
    expect(recoverQaArgs(["6"], { npm_config_count: "true" })).toEqual(["--count", "6"]);
    // Two orphaned value flags cannot be attributed safely.
    expect(() =>
      recoverQaArgs(["6", "7"], { npm_config_count: "true", npm_config_seed_base: "true" }),
    ).toThrow(/equals form/);
    // Explicit flags are never duplicated by recovery.
    expect(recoverQaArgs(["--count=3"], { npm_config_count: "3" })).toEqual(["--count=3"]);
  });

  it("mulberry32 is a stable deterministic sequence in [0, 1)", () => {
    const rng = mulberry32(123);
    const seq = [rng(), rng(), rng()];
    const rng2 = mulberry32(123);
    expect([rng2(), rng2(), rng2()]).toEqual(seq);
    for (const v of seq) {
      expect(v).toBeGreaterThanOrEqual(0);
      expect(v).toBeLessThan(1);
    }
  });
});

describe("qa exit interview extraction", () => {
  it("parses the single tagged fence", () => {
    const text = 'log…\n```json exit-interview\n{"clarity": 3}\n```\n';
    expect(extractExitInterview(text)).toEqual({ ok: true, value: { clarity: 3 } });
  });

  it("rejects a plain json fence, missing fences, and duplicates", () => {
    expect(extractExitInterview('```json\n{"clarity": 3}\n```').ok).toBe(false);
    expect(extractExitInterview("no fence here").ok).toBe(false);
    const twice = '```json exit-interview\n{"a":1}\n```\n```json exit-interview\n{"b":2}\n```';
    expect(extractExitInterview(twice).ok).toBe(false);
  });

  it("rejects invalid JSON and non-object payloads", () => {
    expect(extractExitInterview("```json exit-interview\nnot json\n```").ok).toBe(false);
    expect(extractExitInterview("```json exit-interview\n[1]\n```").ok).toBe(false);
  });
});

describe("qa report summary", () => {
  it("aggregates ratings, severities, per-persona stats, and failures", () => {
    const entries = [
      {
        seed: 1,
        persona: "critic",
        tier: "cheap",
        model: QA_TIERS.cheap.model,
        interview: {
          clarity: 2,
          enjoyment: 3,
          would_replay: false,
          got_stuck: true,
          confusions: ["opening text is dense"],
          bugs: [{ where: "station", severity: "S3", note: "blocked door" }],
        },
        error: null,
      },
      {
        seed: 2,
        persona: "casual",
        tier: "cheap",
        model: QA_TIERS.cheap.model,
        interview: {
          clarity: 4,
          enjoyment: 4,
          would_replay: true,
          got_stuck: false,
          confusions: [],
          bugs: [],
        },
        error: null,
      },
      {
        seed: 3,
        persona: "critic",
        tier: "think",
        model: QA_TIERS.think.model,
        interview: null,
        error: "codex exited with status 1",
      },
    ];
    const summary = summarizeQaReports(entries);
    expect(summary.total).toBe(3);
    expect(summary.reported).toBe(2);
    expect(summary.failed).toBe(1);
    expect(summary.clarity_total).toBe(6);
    expect(summary.enjoyment_total).toBe(7);
    expect(summary.would_replay).toBe(1);
    expect(summary.got_stuck).toBe(1);
    expect(summary.bugs_by_severity.S3).toBe(1);
    expect(summary.by_persona.critic.runs).toBe(2);
    expect(summary.by_persona.critic.reported).toBe(1);
    expect(summary.by_persona.casual.zero_negative).toBe(1);
    // Findings sort most-severe first; confusions rank below S0.
    expect(summary.findings[0].severity).toBe("S3");
    expect(summary.findings[summary.findings.length - 1].severity).toBe("confusion");
    expect(summary.failures).toEqual([
      { seed: 3, persona: "critic", error: "codex exited with status 1" },
    ]);
  });
});
